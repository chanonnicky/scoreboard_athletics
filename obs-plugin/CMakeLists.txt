cmake_minimum_required(VERSION 3.28...3.30)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/common/bootstrap.cmake" NO_POLICY_SCOPE)
project(${_name} VERSION ${_version} LANGUAGES C CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/buildspec.json")
set(ENABLE_FRONTEND_API ON CACHE BOOL "OBS dock support" FORCE)
set(ENABLE_QT ON CACHE BOOL "Native Qt renderer" FORCE)
include(compilerconfig)
include(defaults)
include(helpers)
find_package(libobs REQUIRED)
find_package(obs-frontend-api REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
add_library(cg-core STATIC src/model.cpp src/model.hpp src/renderer.cpp src/renderer.hpp)
target_include_directories(cg-core PUBLIC src)
set_target_properties(cg-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(cg-core PUBLIC Qt6::Core Qt6::Gui)
add_library(${CMAKE_PROJECT_NAME} MODULE src/plugin.cpp src/dock.cpp src/dock.hpp)
target_link_libraries(${CMAKE_PROJECT_NAME} PRIVATE cg-core Qt6::Widgets OBS::libobs OBS::obs-frontend-api)
if(MSVC)
  target_compile_options(cg-core PRIVATE /utf-8)
  target_compile_options(${CMAKE_PROJECT_NAME} PRIVATE /utf-8)
endif()
set_target_properties_plugin(${CMAKE_PROJECT_NAME} PROPERTIES OUTPUT_NAME ${_name})
include(CTest)
if(BUILD_TESTING)
  add_executable(cg-tests tests/core-test.cpp)
  target_link_libraries(cg-tests PRIVATE cg-core Qt6::Widgets)
  if(MSVC)
    target_compile_options(cg-tests PRIVATE /utf-8)
  endif()
  add_test(NAME native-core COMMAND cg-tests)
  add_executable(cg-dock-tests tests/dock-test.cpp src/dock.cpp src/dock.hpp)
  target_link_libraries(cg-dock-tests PRIVATE cg-core Qt6::Widgets)
  if(MSVC)
    target_compile_options(cg-dock-tests PRIVATE /utf-8)
  endif()
  add_test(NAME native-dock COMMAND cg-dock-tests)
  # Optional GPU integration harness; run against an actual OBS runtime, not SDK DLLs.
  add_library(cg-native-smoke MODULE src/plugin.cpp src/dock.cpp src/dock.hpp)
  target_compile_definitions(cg-native-smoke PRIVATE CG_NATIVE_TEST)
  target_link_libraries(cg-native-smoke PRIVATE cg-core Qt6::Widgets OBS::libobs OBS::obs-frontend-api)
  set_target_properties(cg-native-smoke PROPERTIES PREFIX "")
  add_executable(cg-obs-tests tests/obs-test.cpp)
  target_link_libraries(cg-obs-tests PRIVATE cg-core Qt6::Widgets OBS::libobs)
  if(MSVC)
    target_compile_options(cg-native-smoke PRIVATE /utf-8)
    target_compile_options(cg-obs-tests PRIVATE /utf-8)
  endif()
  # OBS's Qt package ships the native platform plugins, not qoffscreen.
  if(WIN32)
    set(CG_TEST_PLATFORM windows)
  else()
    set(CG_TEST_PLATFORM cocoa)
  endif()
  set_tests_properties(native-core native-dock PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=${CG_TEST_PLATFORM};CG_FONT_DIR=${CMAKE_CURRENT_SOURCE_DIR}/data/fonts")
endif()
